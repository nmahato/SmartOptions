def test_get_quote_success(self, mock_cache_set, mock_cache_get, mock_requests):
    print(f"Testing with API key: {self.api_key}")
    result = self.service.get_quote('AAPL')
    print(f"Result: {result}")
    self.assertIsNotNone(result)
